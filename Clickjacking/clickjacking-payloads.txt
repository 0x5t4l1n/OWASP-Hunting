# Header checks to run against the target before building a PoC
curl -sI https://TARGET_HOST/ | grep -i "x-frame-options"
curl -sI https://TARGET_HOST/ | grep -i "content-security-policy"

# Weak/missing header values that still allow framing
X-Frame-Options: ALLOWALL
# (non-standard / ignored by most browsers — functionally equivalent to missing)
Content-Security-Policy: frame-ancestors *
Content-Security-Policy: frame-ancestors 'self' https://attacker.example.com
# (misconfigured allowlist that includes an attacker-controlled or overly broad origin)

# Client-side-only framebuster that can be neutralized via the sandbox attribute
<iframe src="TARGET_URL" sandbox="allow-forms allow-scripts"></iframe>
# sandbox without allow-top-navigation prevents the framed page's own
# "if (top !== self) top.location = self.location;" framebuster from executing
